local function checkVector3()
  local v = Vector3.new(3, 4, 0)
  assert(typeof(v) == "Vector3")
  assert(v.X == 3 and v.Y == 4 and v.Z == 0)
  assert(math.abs(v.Magnitude - 5) < 1e-6)

  local u = v.Unit
  assert(math.abs(u.Magnitude - 1) < 1e-6)
  assert(math.abs(u.X - 0.6) < 0.0001)
  assert(math.abs(u.Y - 0.8) < 0.0001)

  assert(typeof(Vector3.zero) == "Vector3")
  assert(Vector3.zero.X == 0 and Vector3.zero.Y == 0 and Vector3.zero.Z == 0)
  assert(typeof(Vector3.one) == "Vector3")
  assert(Vector3.one.X == 1 and Vector3.one.Y == 1 and Vector3.one.Z == 1)
  assert(typeof(Vector3.xAxis) == "Vector3")
  assert(typeof(Vector3.yAxis) == "Vector3")
  assert(typeof(Vector3.zAxis) == "Vector3")

  assert(Vector3.xAxis:Dot(Vector3.yAxis) == 0)
  assert(Vector3.xAxis:Dot(Vector3.xAxis) == 1)

  local c = Vector3.xAxis:Cross(Vector3.yAxis)
  assert(math.abs(c.X) < 1e-6 and math.abs(c.Y) < 1e-6)
  assert(math.abs(c.Z - 1) < 1e-6)
  assert(math.abs(Vector3.zero:Lerp(Vector3.one, 0.5).X - 0.5) < 1e-6)

  assert(v:FuzzyEq(Vector3.new(3.0000001, 4, 0), 1e-5))
  assert(not v:FuzzyEq(Vector3.new(3.1, 4, 0), 1e-5))
end

local function checkUtf8()
  assert(type(utf8) == "table")
  assert(type(utf8.char) == "function")
  assert(type(utf8.charpattern) == "string" and #utf8.charpattern > 0)
  assert(type(utf8.codepoint) == "function")
  assert(type(utf8.codes) == "function")
  assert(type(utf8.len) == "function")
  assert(type(utf8.offset) == "function")

  assert(utf8.char(65) == "A")
  assert(utf8.char(97) == "a")
  assert(utf8.char(48) == "0")

  local snowman = utf8.char(9731)
  assert(type(snowman) == "string" and #snowman > 1)
  local star = utf8.char(9733)
  assert(type(star) == "string" and #star > 1)
  assert(utf8.codepoint(snowman, 1) == 9731)

  local hello = utf8.char(72, 101, 108, 108, 111)
  local codes = {}
  for _, cp in utf8.codes(hello) do
    table.insert(codes, cp)
  end
  assert(#codes == 5)
  assert(codes[1] == 72 and codes[5] == 111)

  assert(utf8.len("_-_2_") == 5)
  assert(utf8.len("") == 0)

  local mixed = utf8.char(9731, 65, 9733)
  assert(utf8.len(mixed) == 3)
  local off = utf8.offset(mixed, 2)
  assert(type(off) == "number" and off > 1)

  local chars = {}
  for ch in mixed:gmatch(utf8.charpattern) do
    table.insert(chars, ch)
  end
  assert(#chars == 3)
end
